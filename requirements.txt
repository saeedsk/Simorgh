# Everything else is stdlib-only. google-genai is an optional dependency:
# GeminiProvider (src/cognition/gemini_provider.py) imports it lazily, only
# when actually calling the API, so the rest of Simorgh runs with nothing
# installed. Install this only if you're setting GEMINI_API_KEY.
google-genai

# ClaudeCodeProvider (src/cognition/claude_code_provider.py) needs no pip
# package -- it shells out to the `claude` CLI binary, installed
# separately (npm install -g @anthropic-ai/claude-code, or see
# claude.com/product/claude-code) and authenticated once via `claude
# login`. If that binary isn't on PATH, this provider is simply
# unavailable and Simorgh falls back exactly as if no provider existed.

# The interactive prompt (simorgh/interface/tui.py): sticky footer, a
# completion menu for `/` commands and `@` paths, multi-line editing,
# live syntax colouring, persistent history. Optional in the same way as
# the above -- `tui.available()` is false without it and the Interface
# falls back to its readline REPL, which is still the path used whenever
# stdout is not a terminal.
prompt-toolkit>=3.0

# PDF text extraction for `read_file` and `web_fetch`. Optional: both
# tools fall back to pdfminer.six and then to an explicit refusal, so
# nothing here is required to boot -- but GAIA attaches documents, and
# the creator's papers/ directory is PDFs.
pypdf>=6.0
