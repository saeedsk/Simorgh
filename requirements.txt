# Everything else is stdlib-only. google-genai is an optional dependency:
# GeminiProvider (src/cognition/gemini_provider.py) imports it lazily, only
# when actually calling the API, so the rest of Simorgh runs with nothing
# installed. Install this only if you're setting GEMINI_API_KEY.
google-genai

# ClaudeCodeProvider (src/cognition/claude_code_provider.py) needs no pip
# package -- it shells out to the `claude` CLI binary, installed
# separately (npm install -g @anthropic-ai/claude-code, or see
# claude.com/product/claude-code) and authenticated once via `claude
# login`. If that binary isn't on PATH, this provider is simply
# unavailable and Simorgh falls back exactly as if no provider existed.

# The interactive prompt (simorgh/interface/tui.py): sticky footer, a
# completion menu for `/` commands and `@` paths, multi-line editing,
# live syntax colouring, persistent history. Optional in the same way as
# the above -- `tui.available()` is false without it and the Interface
# falls back to its readline REPL, which is still the path used whenever
# stdout is not a terminal.
prompt-toolkit>=3.0

# PDF text extraction for `read_file` and `web_fetch`. Optional: both
# tools fall back to pdfminer.six and then to an explicit refusal, so
# nothing here is required to boot -- but GAIA attaches documents, and
# the creator's papers/ directory is PDFs.
pypdf>=6.0

# Test-only. `pytest -n auto` runs the suite across every core instead
# of one: 3066 tests went from 250s to 57s on a 12-core machine, same
# pass/fail result. Not needed to boot Simorgh itself -- only to run
# its own test suite faster (2026-09-08).
pytest-xdist>=3.6

# `search_listings` (simorgh/execution/realestate.py): real, current
# for-sale property data via Realtor.com's own site backend -- an
# actively-maintained open-source library, not a licensed data API, so
# it can break or get rate-limited without warning. Optional: imported
# lazily, only when the tool is actually called, so nothing here is
# required to boot.
homeharvest>=0.4

# Formats `read_file` and `web_fetch` understand beyond text and PDF
# (simorgh/execution/doctext.py): Word documents, spreadsheets, and an
# image's dimensions/EXIF. All optional and imported lazily -- without
# them those files report what to install, rather than binary noise.
python-docx>=1.1
openpyxl>=3.1
Pillow>=10.0

# Reading the text inside an image. Needs BOTH this and a `tesseract`
# binary on PATH; without either, an image still reports its dimensions
# and says which piece is missing rather than failing.
pytesseract>=0.3

# `notify`'s `apprise` provider (simorgh/execution/notify.py): one
# `APPRISE_URLS` variable covers Telegram, Discord, Pushover, Signal and
# ~100 more services, each written as a URL. Optional and imported
# lazily: with the package absent, `auto` simply passes over the
# provider and naming it explicitly refuses with this pip line. The
# four self-hosted providers beside it (ntfy, Gotify, Home Assistant,
# Matrix) are plain HTTP and need nothing installed at all.
apprise>=1.9
